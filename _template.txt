<script>
const DATA = __PAYLOAD__;
const R = DATA.records, META = DATA.meta, COST = DATA.cost;
const OUT = new Set(["No match","Excluded"]);
const inScope = R.filter(r=>!OUT.has(r.coverage));
const N = inScope.length;

// ---- aggregates ----
function countBy(arr,key){const m={};arr.forEach(r=>{const v=r[key]||"(blank)";m[v]=(m[v]||0)+1});return m;}
const cov = countBy(R,"coverage");
const acquired = (cov["Delivered and verified"]||0)+(cov["Validated full coverage"]||0)+(cov["Validated partial coverage"]||0);
const path = cov["Confirmed acquisition path"]||0;
const excluded = cov["Excluded"]||0;
const nomatch = cov["No match"]||0;
const highP = R.filter(r=>r.priority==="High").length;

// ---- lede ----
document.getElementById("lede").innerHTML =
 `<b>${META.agencies}</b> licensing agencies across <b>${META.jurisdictions}</b> jurisdictions have been cataloged. `+
 `Of the <b>${N}</b> in scope, <b>${acquired}</b> (<b>${Math.round(acquired/N*100)}%</b>) have data acquired or validated; the remaining <b>${path}</b> all have a confirmed acquisition path, so there are <b>no dead ends</b> &mdash; every in-scope agency has at least a known route to the data. `+
 `Nearly all of it is free; only <b>${COST.paidAgencies}</b> agencies carry a fee, most nominal. `+
 `The gating item now is not discovery &mdash; it is the <b>${Object.values(decMapPreview()).reduce((a,b)=>a+b,0)}</b> inclusion, fee and records-request approvals awaiting leadership, shown below.`;
function decMapPreview(){const m={};inScope.forEach(r=>{if(r.decision&&!/no action/i.test(r.decision))m[r.decision]=1+(m[r.decision]||0)});return m;}

// ---- tiles ----
const tiles=[
 {k:META.jurisdictions,l:"Jurisdictions covered",n:"states, DC & territories",accent:true},
 {k:META.agencies,l:"Agencies cataloged",n:`${N} in scope · ${nomatch} unmatched`},
 {k:acquired,sub:`/${N}`,l:"Data acquired or validated",n:`${Math.round(acquired/N*100)}% of in-scope`,accent:true},
 {k:path,l:"Acquisition path confirmed",n:"known route, no dead ends"},
 {k:COST.paidAgencies,l:"Agencies with a fee",n:`${N-COST.paidAgencies} free; fees mostly nominal`},
 {k:highP,l:"High-priority agencies",n:`${META.agencies-highP} medium`},
];
document.getElementById("tiles").innerHTML = tiles.map(t=>
 `<div class="tile ${t.accent?'accent':''}"><div class="k">${t.k}${t.sub?`<small>${t.sub}</small>`:''}</div>`+
 `<div class="l">${t.l}</div><div class="n">${t.n}</div></div>`).join("");

// ---- pipeline ----
const pipeSegs=[
 {n:"Acquired / validated",v:acquired,c:"var(--blue-d)"},
 {n:"Path confirmed, not yet pulled",v:path,c:"var(--warn)"},
];
const pipeTot=acquired+path;
document.getElementById("pipe").innerHTML=pipeSegs.map(s=>
 `<div title="${s.n}: ${s.v}" style="width:${s.v/pipeTot*100}%;background:${s.c}">${s.v}</div>`).join("");
document.getElementById("pipeLegend").innerHTML=pipeSegs.map(s=>
 `<span><i class="sw" style="background:${s.c}"></i>${s.n} &middot; ${s.v}</span>`).join("")+
 `<span style="color:var(--muted)">In-scope: ${N} agencies &nbsp;·&nbsp; out of scope: ${nomatch} unmatched + ${excluded} excluded</span>`;

// ---- bar helper ----
function bars(el,rows,max,color){
 document.getElementById(el).innerHTML=rows.map(r=>
  `<div class="bar"><div class="name" title="${r.n}">${r.n}</div>`+
  `<div class="track"><div class="fill" style="width:${r.v/max*100}%;background:${r.c||color}"></div></div>`+
  `<div class="val">${r.v}</div></div>`).join("");
}
const covColors={"Delivered and verified":"var(--good)","Validated full coverage":"var(--blue)",
 "Validated partial coverage":"var(--blue-m)","Confirmed acquisition path":"var(--warn)",
 "Excluded":"var(--gray)","No match":"var(--gray-l)"};
const covOrder=["Delivered and verified","Validated full coverage","Validated partial coverage","Confirmed acquisition path","Excluded","No match"];
bars("coverageBars",covOrder.filter(k=>cov[k]).map(k=>({n:k,v:cov[k],c:covColors[k]})),Math.max(...Object.values(cov)));

const acc=countBy(R,"access");delete acc["No match"];delete acc["Not applicable"];
const accRows=Object.entries(acc).sort((a,b)=>b[1]-a[1]).map(([n,v])=>({n,v}));
bars("accessBars",accRows,Math.max(...accRows.map(r=>r.v)),"var(--blue)");

// ---- cost ----
const paidC=countBy(inScope,"paid");
const costRows=[["Free / no cost",paidC["No"]||0],["Paid",paidC["Yes"]||0],["Optional fee",paidC["Optional"]||0],["Unknown / pending",paidC["Unknown"]||0]]
 .map(([n,v])=>({n,v}));
bars("costBars",costRows,Math.max(...costRows.map(r=>r.v)),"var(--blue)");
document.getElementById("costNote").innerHTML=
 `Known flat fees total ≈ <b>$${COST.oneTimeFloor.toLocaleString()}</b> across ${COST.withStatedFee} agencies (a floor). Several fees are priced <b>per list / per record / per roster</b>, so recurring cost depends on pull cadence and is not yet fully modeled.`;

// ---- decisions ----
const decMap={};inScope.forEach(r=>{const d=r.decision;if(!d)return;
 let bucket="Other";
 if(/state fee/i.test(d))bucket="Approve inclusion + state fee";
 else if(/records request/i.test(d))bucket="Authorize official records request";
 else if(/confirm inclusion/i.test(d))bucket="Confirm inclusion";
 else if(/accept validated population/i.test(d))bucket="Accept population or request expansion";
 else if(/no action/i.test(d))bucket="No action needed";
 decMap[bucket]=(decMap[bucket]||0)+1;});
const decEx={"Approve inclusion + state fee":"Green-light paid agencies so ordering can begin",
 "Authorize official records request":"Approve the formal request to the agency",
 "Confirm inclusion":"Confirm the agency belongs in the registry",
 "Accept population or request expansion":"Accept the validated set or ask for more",
 "No action needed":"Scope unchanged"};
const decOrder=["Approve inclusion + state fee","Authorize official records request","Confirm inclusion","Accept population or request expansion","No action needed","Other"];
document.getElementById("decisions").innerHTML=decOrder.filter(k=>decMap[k]).map(k=>
 `<div class="dec"><div class="cnt">${decMap[k]}</div><div class="txt">${k}<small>${decEx[k]||""}</small></div></div>`).join("");

// ---- table ----
const covPill={"Delivered and verified":"st-del","Validated full coverage":"st-full",
 "Validated partial coverage":"st-part","Confirmed acquisition path":"st-path",
 "Excluded":"st-out","No match":"st-out"};
let sortK="state",sortDir=1;
function render(){
 const q=document.getElementById("search").value.toLowerCase();
 let rows=R.filter(r=>!q||[r.state,r.agency,r.coverage,r.access,r.paid,r.fee,r.decision,r.priority].join(" ").toLowerCase().includes(q));
 rows.sort((a,b)=>{const x=(a[sortK]||"").toString(),y=(b[sortK]||"").toString();return x<y?-sortDir:x>y?sortDir:0;});
 document.getElementById("tbody").innerHTML=rows.map(r=>
  `<tr><td><b>${r.state}</b></td><td>${r.agency||"&mdash;"}</td>`+
  `<td class="${r.priority==='High'?'prioH':''}">${r.priority||""}</td>`+
  `<td><span class="pill ${covPill[r.coverage]||'st-out'}">${r.coverage||"&mdash;"}</span></td>`+
  `<td>${r.access||"&mdash;"}</td><td class="${r.paid==='Yes'?'paidY':''}">${r.paid||"&mdash;"}</td>`+
  `<td class="wrap">${r.fee&&r.fee!=='No agency fee identified'&&r.fee!=='No match'?r.fee:'<span style="color:var(--muted)">&mdash;</span>'}</td>`+
  `<td class="wrap">${r.decision&&r.decision!=='No match'?r.decision:'<span style="color:var(--muted)">&mdash;</span>'}</td></tr>`).join("");
 document.getElementById("tblCount").textContent=`${rows.length} of ${R.length} agencies`;
}
document.querySelectorAll("th").forEach(th=>th.addEventListener("click",()=>{
 const k=th.dataset.k;if(k===sortK)sortDir*=-1;else{sortK=k;sortDir=1;}render();}));
document.getElementById("search").addEventListener("input",render);
render();

// ---- footer ----
document.getElementById("foot").innerHTML=
 `<b>Source:</b> Lighthouse_API_Master_Discovery_Tracker (API Discovery Tracker sheet), compiled by M. Murtuza. `+
 `<b>Denominator:</b> ${N} in-scope agencies; ${nomatch} agencies returned no pharmacy-relevant match and are excluded from percentages. `+
 `<b>Coverage status</b> is read from the enriched "Coverage – Licensing Coverage" field, not the stale "Discovery Status" column. `+
 `<b>Cost</b> figures are a floor: only clearly-stated flat fees are summed; per-list/per-record pricing and pending agency quotes are not modeled. `+
 `Not yet reflected: weighting by LighthouseAI's current-customer footprint (needs the customer trading-partner list). `+
 `This is a snapshot; wire the SharePoint source for automatic daily refresh.`;
