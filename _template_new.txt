<script>
const DATA = __PAYLOAD__;
const R = DATA.records, META = DATA.meta;
const yn = v => { v=(v||"").trim().toLowerCase();
  if(["y","yes","true"].includes(v)) return "Yes";
  if(["n","no","false"].includes(v)) return "No";
  return v ? v : "(blank)"; };
function countBy(arr,key,norm){const m={};arr.forEach(r=>{let v=(r[key]||"").trim()||"(blank)";if(norm)v=norm(r[key]);m[v]=(m[v]||0)+1});return m;}
function bars(el,rows,color){
  const max=Math.max(1,...rows.map(r=>r.v));
  document.getElementById(el).innerHTML = rows.length ? rows.map(r=>
    `<div class="bar"><div class="name" title="${r.n}">${r.n}</div>`+
    `<div class="track"><div class="fill" style="width:${r.v/max*100}%;background:${r.c||color}"></div></div>`+
    `<div class="val">${r.v}</div></div>`).join("") : `<div class="cap">No data yet.</div>`;
}
function ynRows(map){ // fixed order Yes / No / blank, with status colors
  const order=[["Yes","var(--good)"],["No","var(--gray)"],["(blank)","var(--gray-l)"]];
  return order.filter(([k])=>map[k]).map(([k,c])=>({n:k,v:map[k],c}));
}

const N = R.length;
const states = new Set(R.map(r=>r.state).filter(Boolean)).size;
const pub = countBy(R,"public",yn);
const cost = countBy(R,"cost",yn);
const herm = countBy(R,"hermai",yn);
const populated = R.filter(r=>[r.public,r.cost,r.hermai,r.mechanism,r.outcome].some(x=>(x||"").trim())).length;

document.getElementById("lede").innerHTML =
  (populated < N || N<=1
    ? `<b>Template preview.</b> ${populated} of ${N} agency ${N===1?"row is":"rows are"} populated so far &mdash; charts fill in as the tracker is completed. `
    : "") +
  `Tracking <b>${N}</b> agencies across <b>${states}</b> ${states===1?"jurisdiction":"jurisdictions"}. `+
  `Publicly available: <b>${pub["Yes"]||0}</b> &middot; on Hermai: <b>${herm["Yes"]||0}</b> &middot; carry a fee: <b>${cost["Yes"]||0}</b>.`;

const tiles=[
  {k:states,l:"Jurisdictions",n:"distinct states",accent:true},
  {k:N,l:"Agencies tracked",n:`${populated} populated`},
  {k:pub["Yes"]||0,l:"Publicly available",n:`${pub["No"]||0} not public`,accent:true},
  {k:cost["Yes"]||0,l:"Carry a fee",n:`${cost["No"]||0} free`},
  {k:herm["Yes"]||0,l:"On Hermai",n:`${herm["No"]||0} not on Hermai`},
];
document.getElementById("tiles").innerHTML = tiles.map(t=>
  `<div class="tile ${t.accent?'accent':''}"><div class="k">${t.k}</div>`+
  `<div class="l">${t.l}</div><div class="n">${t.n}</div></div>`).join("");

bars("publicBars", ynRows(pub));
bars("hermaiBars", ynRows(herm));
bars("costBars", ynRows(cost));
bars("statusBars", ynRows(countBy(R,"inclStatus",yn)));
bars("discBars", ynRows(countBy(R,"inclDiscipline",yn)));

const mech = countBy(R,"mechanism");
const mechRows = Object.entries(mech).filter(([k])=>k!=="(blank)").sort((a,b)=>b[1]-a[1]).map(([n,v])=>({n,v}));
bars("mechBars", mechRows, "var(--blue)");

const feeRows = R.filter(r=>(r.fee||"").toString().trim());
document.getElementById("costNote").innerHTML =
  `<b>${cost["Yes"]||0}</b> agencies carry a fee; <b>${feeRows.length}</b> have a fee amount recorded. Recurring cost depends on pull cadence.`;

// table
let sortK="state",sortDir=1;
function render(){
  const q=document.getElementById("search").value.toLowerCase();
  let rows=R.filter(r=>!q||[r.state,r.agency,r.mechanism,r.outcome,r.public,r.cost,r.hermai].join(" ").toLowerCase().includes(q));
  rows.sort((a,b)=>{const x=(a[sortK]||"").toString(),y=(b[sortK]||"").toString();return x<y?-sortDir:x>y?sortDir:0;});
  const cell=v=>v&&v.toString().trim()?v:'<span style="color:var(--muted)">&mdash;</span>';
  document.getElementById("tbody").innerHTML=rows.map(r=>
    `<tr><td><b>${cell(r.state)}</b></td><td>${cell(r.agency)}</td>`+
    `<td>${cell(yn(r.public))}</td><td>${cell(r.mechanism)}</td>`+
    `<td class="${yn(r.cost)==='Yes'?'paidY':''}">${cell(yn(r.cost))}</td><td class="wrap">${cell(r.fee)}</td>`+
    `<td>${cell(yn(r.hermai))}</td><td>${cell(yn(r.inclStatus))}</td>`+
    `<td>${cell(yn(r.inclDiscipline))}</td><td class="wrap">${cell(r.outcome)}</td></tr>`).join("");
  document.getElementById("tblCount").textContent=`${rows.length} of ${R.length} agencies`;
}
document.querySelectorAll("th").forEach(th=>th.addEventListener("click",()=>{
  const k=th.dataset.k;if(k===sortK)sortDir*=-1;else{sortK=k;sortDir=1;}render();}));
document.getElementById("search").addEventListener("input",render);
render();

document.getElementById("foot").innerHTML=
  `<b>Source:</b> License Data Tracker (simplified schema). <b>Schema:</b> ${META.schema}. `+
  `Y/N fields are normalized (Yes/No/blank). "Outcome" is shown verbatim pending a defined value set. `+
  `Layout will be tuned to real distributions once the tracker is fully populated.`;

  const r=document.documentElement;const cur=r.getAttribute("data-theme");
  const dark=cur?cur==="dark":matchMedia("(prefers-color-scheme: dark)").matches;
  r.setAttribute("data-theme",dark?"light":"dark");});
