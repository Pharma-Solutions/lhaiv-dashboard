const DATA = __PAYLOAD__;
const R = DATA.records, META = DATA.meta;
const yn = v => { v=(v||"").trim().toLowerCase();
  if(["y","yes","true"].includes(v)) return "Yes";
  if(["n","no","false"].includes(v)) return "No";
  return v ? v : "(blank)"; };
function countBy(arr,key,norm){const m={};arr.forEach(r=>{let v=norm?norm(r[key]):((r[key]||"").trim()||"(blank)");m[v]=(m[v]||0)+1});return m;}
function bars(el,rows,color){
  const max=Math.max(1,...rows.map(r=>r.v));
  document.getElementById(el).innerHTML = rows.length ? rows.map(r=>
    `<div class="bar"><div class="name" title="${r.n}">${r.n}</div>`+
    `<div class="track"><div class="fill" style="width:${r.v/max*100}%;background:${r.c||color}"></div></div>`+
    `<div class="val">${r.v}</div></div>`).join("") : `<div class="cap">No data yet.</div>`;
}

const OUT = "Out of Scope";
const inScope = R.filter(r => (r.outcome||"").trim() !== OUT);
const N = inScope.length;
const outScope = R.length - N;
const obtained = inScope.filter(r => (r.outcome||"").toLowerCase().startsWith("obtained")).length;
const notObtained = inScope.filter(r => (r.outcome||"").toLowerCase().startsWith("not obtained")).length;
const herm = countBy(inScope, "hermai", yn);
const cost = countBy(inScope, "cost", yn);
const pub  = countBy(inScope, "public", yn);

document.getElementById("lede").innerHTML =
  `<b>${R.length}</b> agency records &mdash; <b>${N}</b> in scope, <b>${outScope}</b> out of scope. `+
  `<b>${obtained}</b> of ${N} in-scope agencies have data <b>obtained</b>; <b>${notObtained}</b> not yet. `+
  `Publicly available: <b>${pub["Yes"]||0}</b> &middot; carry a fee: <b>${cost["Yes"]||0}</b> &middot; `+
  `still flagged for <b>Hermai</b>: <b>${herm["Yes"]||0}</b> (the number to drive toward zero).`;

const tiles=[
  {k:N,l:"Agencies in scope",n:`${outScope} out of scope`,accent:true},
  {k:obtained,sub:`/${N}`,l:"Data obtained",n:`${Math.round(obtained/Math.max(1,N)*100)}% of in-scope`,accent:true},
  {k:notObtained,l:"Not yet obtained",n:"acquisition pending"},
  {k:herm["Yes"]||0,l:"Hermai-flagged",n:`${herm["No"]||0} without Hermai`},
  {k:cost["Yes"]||0,l:"Carry a fee",n:`${cost["No"]||0} free / no cost`},
  {k:pub["Yes"]||0,l:"Publicly available",n:`${pub["No"]||0} not public`},
];
document.getElementById("tiles").innerHTML = tiles.map(t=>
  `<div class="tile ${t.accent?'accent':''}"><div class="k">${t.k}${t.sub?`<small>${t.sub}</small>`:''}</div>`+
  `<div class="l">${t.l}</div><div class="n">${t.n}</div></div>`).join("");

// outcome ladder (in-scope, ordered)
const oc = countBy(inScope, "outcome");
const ocOrder = ["Obtained - Free","Obtained - Purchased","Obtained","Partially Obtained - Still working","Not Obtained"];
const ocColor = {"Obtained - Free":"var(--good)","Obtained - Purchased":"var(--blue)","Obtained":"var(--blue-m)",
                 "Partially Obtained - Still working":"var(--warn)","Not Obtained":"var(--gray)"};
const ocRows = ocOrder.filter(k=>oc[k]).map(k=>({n:k,v:oc[k],c:ocColor[k]}));
Object.keys(oc).forEach(k=>{ if(!ocOrder.includes(k) && k!=="(blank)") ocRows.push({n:k,v:oc[k],c:"var(--gray)"}); });
bars("outcomeBars", ocRows);

const mech = countBy(inScope, "mechanism");
delete mech["(blank)"]; delete mech["Not Applicable"];
bars("mechBars", Object.entries(mech).sort((a,b)=>b[1]-a[1]).map(([n,v])=>({n,v})), "var(--blue)");

const ynRows = (m)=>[["Yes","var(--warn)"],["No","var(--good)"],["(blank)","var(--gray-l)"],["Unknown","var(--gray)"]]
  .filter(([k])=>m[k]).map(([k,c])=>({n:k,v:m[k],c}));
bars("costBars", [["Free / no cost", cost["No"]||0,"var(--good)"],["Paid", cost["Yes"]||0,"var(--warn)"],
                  ["Unknown", cost["Unknown"]||0,"var(--gray)"]].filter(r=>r[1]).map(([n,v,c])=>({n,v,c})));
const feeRows = inScope.filter(r=>(r.fee||"").toString().trim() && (r.fee||"").toString().trim()!=="0");
document.getElementById("costNote").innerHTML =
  `<b>${cost["Yes"]||0}</b> agencies carry a fee; <b>${feeRows.length}</b> have a fee amount recorded. Purchases go through the human approval gate.`;
bars("hermaiBars", [["Flagged for Hermai", herm["Yes"]||0,"var(--warn)"],["No Hermai needed", herm["No"]||0,"var(--good)"]]
     .filter(r=>r[1]).map(([n,v,c])=>({n,v,c})));

// table
let sortK="state",sortDir=1;
const ocPill = o => { o=(o||"").toLowerCase();
  if(o.startsWith("obtained")) return "st-del"; if(o.startsWith("partially")) return "st-path";
  if(o.startsWith("not obtained")) return "st-part"; if(o===OUT.toLowerCase()) return "st-out"; return "st-out"; };
function render(){
  const q=document.getElementById("search").value.toLowerCase();
  let rows=R.filter(r=>!q||[r.state,r.agency,r.mechanism,r.outcome,r.cost,r.hermai].join(" ").toLowerCase().includes(q));
  rows.sort((a,b)=>{const x=(a[sortK]||"").toString(),y=(b[sortK]||"").toString();return x<y?-sortDir:x>y?sortDir:0;});
  const cell=v=>v&&v.toString().trim()?v:'<span style="color:var(--muted)">&mdash;</span>';
  document.getElementById("tbody").innerHTML=rows.map(r=>
    `<tr><td><b>${cell(r.state)}</b></td><td>${cell(r.agency)}</td>`+
    `<td>${cell(yn(r.public))}</td><td>${cell(r.mechanism)}</td>`+
    `<td class="${yn(r.cost)==='Yes'?'paidY':''}">${cell(yn(r.cost))}</td><td class="wrap">${cell(r.fee)}</td>`+
    `<td class="${yn(r.hermai)==='Yes'?'paidY':''}">${cell(yn(r.hermai))}</td>`+
    `<td><span class="pill ${ocPill(r.outcome)}">${cell(r.outcome)}</span></td></tr>`).join("");
  document.getElementById("tblCount").textContent=`${rows.length} of ${R.length} agencies`;
}
document.querySelectorAll("th").forEach(th=>th.addEventListener("click",()=>{
  const k=th.dataset.k;if(k===sortK)sortDir*=-1;else{sortK=k;sortDir=1;}render();}));
document.getElementById("search").addEventListener("input",render);
render();

document.getElementById("foot").innerHTML=
  `<b>Source:</b> License Data Tracker (schema: ${META.schema}). In-scope excludes rows marked "Out of Scope". `+
  `"Hermai reliance" counts agencies still flagged Hermai-recommended &mdash; the bulk-path work aims to drive this to zero. `+
  `Y/N fields normalized (Yes/No/blank).`;

  r.setAttribute("data-theme",dark?"light":"dark");});
